# Fixture for the ctest runners: node (wasm-run) and browsers (relink-web.sh + browser-ctest.mjs).
cmake_minimum_required(VERSION 3.20)
project(bwctest CXX)
enable_testing()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_LIBRARY_ARCHITECTURE STREQUAL "wasm64-emscripten")
  message(FATAL_ERROR "toolchain reports ${CMAKE_SIZEOF_VOID_P}-byte pointers / ${CMAKE_LIBRARY_ARCHITECTURE}")
endif()
add_executable(t_basic t.cc)
add_executable(t_file t.cc)
add_executable(t_threads t.cc)
add_executable(t_fail t.cc)
add_executable(t_skip t.cc)
foreach(t basic file threads fail skip)
  target_compile_definitions(t_${t} PRIVATE MODE_${t})
endforeach()
add_test(NAME basic COMMAND t_basic)
add_test(NAME file COMMAND t_file ${CMAKE_CURRENT_SOURCE_DIR}/data/input.txt)
add_test(NAME threads COMMAND t_threads)
add_test(NAME expected_failure COMMAND t_fail)
set_tests_properties(expected_failure PROPERTIES WILL_FAIL TRUE)
add_test(NAME skipped COMMAND t_skip)
set_tests_properties(skipped PROPERTIES SKIP_RETURN_CODE 77)
